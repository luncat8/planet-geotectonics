vec3 speedToHsl(vec2 velocity, float speedContrast) {
    #define INV_TWO_PI 0.15915494309189535

    // Direction normalized to [0, 1):
    // right = 0, top = 0.25, left = 0.5, bottom = 0.75
    float t = fract(atan(velocity.y, velocity.x) * INV_TWO_PI);

    // Hue mapping, piecewise linear between the anchors, one wrap:
    // right = green (120°), top = blue (240°), left = red (0°), bottom = yellow (60°)
    float hue = 120.0 + 480.0 * t;                       // right -> top -> left: 120 -> 240 -> 360
    hue += step(0.5, t) * (120.0 - 240.0 * t);           // left -> bottom -> right: 360 -> 420 -> 480
    hue = mod(hue, 360.0);

    // Speed mapped to lightness. velocity arrives pre-divided by 80000/6 (sim units are
    // m/Myr, 1e4 per cm/yr), so length 6 = the speed view's 8 cm/yr full scale.
    float speed = length(velocity);
    float normalizedSpeed = min(speed * 0.166666667, 1.0);
    float lightness = 0.05 +
        0.65 * pow(normalizedSpeed, speedContrast);

    // HSL:
    // x = hue in degrees
    // y = saturation [0, 1]
    // z = lightness [0, 1]
    return vec3(hue, 0.90, lightness);
}
